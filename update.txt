    echo "Plasma updated → refreshing Wayland components..."
    sudo pacman -S kwin plasma-workspace plasma-desktop qt6-wayland --noconfirm
fi

  # 6) OPTIONAL: RESTART SERVICES (safe)
sudo systemctl daemon-reload
